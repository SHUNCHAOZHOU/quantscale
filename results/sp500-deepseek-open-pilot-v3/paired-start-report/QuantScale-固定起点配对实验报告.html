A dollar-neutral, liquidity-aware cross-sectional equity portfolio can profit from persistent 12-1 and 6-1 momentum while controlling risk by tilting toward lower-volatility names and incorporating a mild short-term reversal adjustment. Long the top-ranked and short the bottom-ranked eligible stocks, equal-weighted on each side, to keep gross exposure at 1.0 and net exposure near zero under the 2% single-name cap.

import numpy as np
import pandas as pd

def generate_weights(history, universe):
    universe = pd.Index(universe)
    if len(universe) == 0:
        return pd.Series(dtype=float)

    adj = history.get('adjusted_close')
    if adj is None:
        adj = history.get('close')
    if adj is None or not isinstance(adj, pd.DataFrame):
        return pd.Series(0.0, index=universe)

    cols = [c for c in universe if c in adj.columns]
    if len(cols) == 0:
        return pd.Series(0.0, index=universe)

    if not adj.index.is_monotonic_increasing:
        adj = adj.sort_index()
    px = adj[cols].tail(320).copy()
    n = len(px)
    if n < 40:
        return pd.Series(0.0, index=universe)

    def _mom(start_lag, end_lag):
        if n <= end_lag + 2:
            return pd.Series(np.nan, index=cols)
        end_pos = n - 1 - end_lag
        start_pos = max(0, n - 1 - start_lag)
        if end_pos <= start_pos:
            return pd.Series(np.nan, index=cols)
        return px.iloc[end_pos] / px.iloc[start_pos] - 1.0

    mom12 = _mom(252, 21)
    mom6 = _mom(126, 21)
    if n > 21:
        ret21 = px.iloc[-1] / px.iloc[n - 1 - 21] - 1.0
    else:
        ret21 = pd.Series(np.nan, index=cols)

    rets = px.pct_change()
    vol_window = min(126, max(2, n - 1))
    vol = rets.tail(vol_window).std() * np.sqrt(252.0)
    vol = vol.replace([np.inf, -np.inf], np.nan)

    last_price = px.iloc[-1]
    valid = last_price.notna() & (last_price > 0) & mom12.notna()
    valid = valid & (px.notna().sum() >= 60)

    if 'eligible' in history:
        edf = history['eligible']
        if isinstance(edf, pd.DataFrame):
            if not edf.index.is_monotonic_increasing:
                edf = edf.sort_index()
            ecols = [c for c in cols if c in edf.columns]
            if ecols:
                e_last = edf[ecols].tail(1)
                if len(e_last) > 0:
                    e = e_last.iloc[-1].reindex(cols).fillna(False).astype(bool)
                    valid = valid & e

    idx_valid = valid[valid].index
    if len(idx_valid) < 10:
        return pd.Series(0.0, index=universe)

    mom6 = mom6.where(mom6.notna(), mom12)
    mom12_v = mom12.loc[idx_valid]
    mom6_v = mom6.loc[idx_valid]
    vol_v = vol.loc[idx_valid].fillna(vol.loc[idx_valid].median())
    if vol_v.isna().all():
        vol_v = pd.Series(1.0, index=idx_valid)
    ret21_v = ret21.loc[idx_valid].fillna(ret21.loc[idx_valid].median())
    if ret21_v.isna().all():
        ret21_v = pd.Series(0.0, index=idx_valid)

    def rp(s):
        return s.rank(method='average', pct=True)

    score = (0.55 * rp(mom12_v) +
             0.25 * rp(mom6_v) +
             0.20 * (1.0 - rp(vol_v)) -
             0.20 * rp(ret21_v))
    score = score.replace([np.inf, -np.inf], np.nan).dropna()
    if len(score) < 10:
        return pd.Series(0.0, index=universe)

    n_side = min(50, len(score) // 2)
    if n_side < 1:
        return pd.Series(0.0, index=universe)

    longs = score.nlargest(n_side).index
    shorts = score.nsmallest(n_side).index
    w = min(0.02, 0.5 / float(n_side))
    weights = pd.Series(0.0, index=universe)
    weights.loc[longs] = w
    weights.loc[shorts] = -w

    gross = weights.abs().sum()
    if gross > 1.0 + 1e-12:
        weights = weights * (1.0 / gross)
    weights = weights.clip(-0.02, 0.02)
    return weights.reindex(universe).fillna(0.0)

初始W2：假设与代码

Cross-sectional short-term reversal, tempered by intermediate momentum and low-volatility preference, produces a market-neutral long-short portfolio. Recent losers with positive medium-term trend and lower realized volatility should outperform recent winners, while rank-based selection and dollar-neutral equal weighting control risk and turnover. Use 10-day reversal, 80-day momentum, 20-day volatility, and select up to 30 names per side.

import numpy as np
import pandas as pd

def _rank_pct(s):
    s = s.dropna()
    if len(s) == 0:
        return s
    return s.rank(method='average', pct=True) - 0.5

def generate_weights(history, universe):
    tickers = pd.Index(list(universe))
    if len(tickers) == 0:
        return pd.Series(dtype=float)

    px = history.get('adjusted_close')
    if px is None:
        return pd.Series(0.0, index=tickers)

    px = px.reindex(columns=tickers).astype(float)
    px = px.replace([np.inf, -np.inf], np.nan)
    if px.shape[0] < 30:
        return pd.Series(0.0, index=tickers)

    last_px = px.iloc[-1]

    ret = px / px.shift(1) - 1.0
    ret = ret.replace([np.inf, -np.inf], np.nan)

    r10 = px / px.shift(10) - 1.0
    r80 = px / px.shift(80) - 1.0
    vol20 = ret.rolling(20, min_periods=15).std()

    last_r10 = r10.iloc[-1]
    last_r80 = r80.iloc[-1]
    last_vol = vol20.iloc[-1]

    valid = (
        last_px.notna()
        & (last_px > 0)
        & last_r10.notna()
        & last_r80.notna()
        & last_vol.notna()
        & (last_vol > 0)
    )

    elig = history.get('eligible')
    if elig is not None:
        elig = elig.reindex(columns=tickers)
        if elig.shape[0] > 0:
            e_last = elig.iloc[-1]
            valid = valid & e_last.fillna(False).astype(bool)

    valid_t = valid[valid].index
    if len(valid_t) < 6:
        return pd.Series(0.0, index=tickers)

    rr10 = _rank_pct(last_r10.reindex(valid_t))
    rr80 = _rank_pct(last_r80.reindex(valid_t))
    rv = _rank_pct(last_vol.reindex(valid_t))

    score = -1.0 * rr10 + 0.55 * rr80 - 0.20 * rv
    score = score.replace([np.inf, -np.inf], np.nan).dropna()
    if len(score) < 4:
        return pd.Series(0.0, index=tickers)

    n_side = min(30, len(score) // 2)
    if n_side < 1:
        return pd.Series(0.0, index=tickers)

    order = score.sort_values(ascending=False)
    longs = order.index[:n_side]
    shorts = order.index[-n_side:]

    w = min(0.02, 0.5 / float(n_side))
    weights = pd.Series(0.0, index=tickers)
    weights.loc[longs] = w
    weights.loc[shorts] = -w

    gross = float(weights.abs().sum())
    if gross > 1.0:
        weights = weights * (1.0 / gross)

    return weights.replace([np.inf, -np.inf], 0.0).fillna(0.0)
初始W3：假设与代码

Liquid equities exhibit short-horizon cross-sectional reversal: recent weak performers rebound relative to recent strong performers over the following week. A dollar-neutral long-short portfolio ranked by negative five-day return scaled by twenty-day volatility, restricted to reasonably liquid names and rebalanced every five sessions, should capture this effect while controlling turnover and idiosyncratic risk.

import numpy as np
import pandas as pd

def generate_weights(history, universe):
    idx = pd.Index(universe)
    if len(idx) == 0:
        return pd.Series(dtype=float)
    zero = pd.Series(0.0, index=idx, dtype=float)

    ac = history.get('adjusted_close')
    if ac is None:
        ac = history.get('close')
    if ac is None:
        return zero

    n = len(ac.index)
    if n < 30:
        return zero

    reb = ((n - 1) // 5) * 5
    if reb < 25:
        return zero

    cols = [c for c in idx if c in ac.columns]
    if len(cols) == 0:
        return zero

    start = max(0, reb - 44)
    close = ac.iloc[start:reb + 1][cols].astype(float).copy()
    close = close.replace([np.inf, -np.inf], np.nan)
    if close.shape[0] < 25:
        return zero

    last = close.iloc[-1]
    valid = last.notna() & (close.notna().sum(axis=0) >= 25)

    ret = close / close.shift(1) - 1.0
    ret = ret.replace([np.inf, -np.inf], np.nan)
    vol = ret.iloc[-20:].std(axis=0, ddof=1)
    vol = vol.where(vol > 0, np.nan)

    r5 = close.iloc[-1] / close.iloc[-6] - 1.0
    eps = 1e-8
    score = -r5 / (vol + eps)

    vol_df = history.get('volume')
    if vol_df is not None:
        vcols = [c for c in cols if c in vol_df.columns]
        if len(vcols) > 0:
            vsub = vol_df.iloc[start:reb + 1][vcols].astype(float)
            csub = close[vcols]
            dollar = vsub * csub
            liq = dollar.iloc[-20:].median(axis=0)

            valid = valid & (liq > 0)

    valid = valid & score.notna()

    shorts = sc.index[-k:].tolist()
    wmag = min(0.5 / k, 0.02)

    w = zero.copy()
    w.loc[longs] = wmag
    w.loc[shorts] = -wmag
    w = w.replace([np.inf, -np.inf], 0.0).fillna(0.0)
    return w

The prior daily short-horizon reversal book had positive gross alpha but was overwhelmed by ~25-33% daily turnover and transaction costs. A monthly-held, cost-aware reversal portfolio should retain much of that gross alpha while cutting turnover by roughly 90%. I will form a dollar-neutral portfolio at 21-trading-day rebalance dates using 21-day and 63-day cumulative-return ranks (long recent losers, short recent winners), restrict to liquid names and exclude the highest-volatility decile, then equal-weight each side under the 2% single-name cap. This tests whether slower implementation, not signal failure, explains the weak net validation performance.

import pandas as pd


def generate_weights(history, universe):

    # signal computed at the most recent rebalance close.

    n = len(px)

Rather than dilute this with momentum or reversal legs that were also negative, the next experiment flips the low-volatility leg: long the top 50 high-volatility eligible/liquid names and short the bottom 50 low-volatility names, dollar-neutral, rebalanced only on a 21-session anchor, capped at 2% per name, with gross exposure at or below 1.0. This isolate-then-invert test should preserve the low turnover and construction of the negative low-vol book while harvesting the opposite sign.

import numpy as np
import pandas as pd


def _get_panel(history, field):
    df = history.get(field)
    if isinstance(df, pd.DataFrame) and not df.empty:
        return df
    return None


def generate_weights(history, universe):
    universe = pd.Index(universe)
    zero = pd.Series(0.0, index=universe, dtype=float)
    if len(universe) == 0:
        return zero

    ac = _get_panel(history, 'adjusted_close')
    if ac is None:
        ac = _get_panel(history, 'close')
    if ac is None:
        return zero

    dates = ac.index
    n = len(dates)
    rebalance = 21
    min_obs = 150
    apos = ((n - 1) // rebalance) * rebalance
    if apos < min_obs:
        return zero
    anchor_date = dates[apos]

    cols = [c for c in universe if c in ac.columns]
    if len(cols) < 8:
        return zero

    raw = ac.loc[:anchor_date, cols]
    prices = raw.ffill()
    if prices.shape[0] < 60:
        return zero

    raw_last = raw.iloc[-1]
    last = prices.iloc[-1]

    elig_df = _get_panel(history, 'eligible')
    if elig_df is not None:
        sub = elig_df.loc[:anchor_date]
        if len(sub) > 0:
            e_raw = sub.iloc[-1].reindex(cols)
            e = e_raw.fillna(False).astype(bool)
        else:
            e = pd.Series(True, index=cols)
    else:
        e = pd.Series(True, index=cols)

    valid = (
        e
        & raw_last.notna()
        & np.isfinite(raw_last.astype(float))
        & (raw_last.astype(float) > 0.0)
        & last.notna()
        & np.isfinite(last.astype(float))
        & (last.astype(float) > 0.0)
    )

    vol_df = _get_panel(history, 'volume')
    close_df = _get_panel(history, 'close')
    if vol_df is not None and close_df is not None:
        v = vol_df.loc[:anchor_date, cols].reindex(index=prices.index, columns=cols)
        px = close_df.loc[:anchor_date, cols].reindex(index=prices.index, columns=cols)
        dollar = (v.astype(float) * px.astype(float)).tail(60)
        with np.errstate(all='ignore'):
            med_dv = dollar.median(axis=0)
        valid = valid & med_dv.notna() & (med_dv >= 1000000.0)

    if int(valid.sum()) < 8:
        return zero

    rets = prices.astype(float).pct_change()
    with np.errstate(all='ignore'):
        vol = rets.rolling(120, min_periods=60).std().iloc[-1]

    f = pd.DataFrame({'vol': vol}).reindex(cols)
    f = f.replace([np.inf, -np.inf], np.nan)
    f = f.loc[valid].dropna(subset=['vol'])
    f = f[f['vol'] > 0.0]
    if len(f) < 8:
        return zero

    vol_rank = f['vol'].rank(pct=True, method='average')
    score = vol_rank.sort_values(ascending=False)

    n_side = min(50, len(score) // 2)
    if n_side < 1:
        return zero

    longs = score.index[:n_side]
    shorts = score.index[-n_side:]
    side = min(0.02, 0.5 / float(n_side))

    out = zero.copy()
    out.loc[longs] = side
    out.loc[shorts] = -side
    out = out.replace([np.inf, -np.inf], 0.0).fillna(0.0)

    gross = float(out.abs().sum())
    if gross > 1.0 and gross > 0.0:
        out = out * (1.0 / gross)

    net = float(out.sum())
    if abs(net) > 0.01:
        active = out[out != 0.0]
        if len(active) > 0:
            adj = net / float(len(active))
            out.loc[active.index] = out.loc[active.index] - adj
            out = out.clip(-0.02, 0.02)
            gross = float(out.abs().sum())
            if gross > 1.0 and gross > 0.0:
                out = out * (1.0 / gross)

    return out.astype(float)
